package com.paytrace.model;

import java.math.BigDecimal;
import java.time.LocalDateTime;

public abstract class BaseFinancialRecord {
    protected String recordId;
    protected LocalDateTime timestamp;
    protected BigDecimal amount;
    protected String currency;

    public BaseFinancialRecord(String recordId, LocalDateTime timestamp, BigDecimal amount, String currency) {
        if (amount != null && amount.compareTo(BigDecimal.ZERO) < 0) {
            throw new IllegalArgumentException("Amount cannot be negative: " + amount);
        }
        this.recordId = recordId;
        this.timestamp = timestamp != null ? timestamp : LocalDateTime.now();
        this.amount = amount != null ? amount : BigDecimal.ZERO;
        this.currency = currency != null ? currency : "USD";
    }

    public String getRecordId() { return recordId; }
    public void setRecordId(String recordId) { this.recordId = recordId; }
    public LocalDateTime getTimestamp() { return timestamp; }
    public void setTimestamp(LocalDateTime timestamp) { this.timestamp = timestamp; }
    public BigDecimal getAmount() { return amount; }
    public void setAmount(BigDecimal amount) {
        if (amount != null && amount.compareTo(BigDecimal.ZERO) < 0) {
            throw new IllegalArgumentException("Amount cannot be negative");
        }
        this.amount = amount;
    }
    public String getCurrency() { return currency; }
    public void setCurrency(String currency) { this.currency = currency; }

    public abstract String getSourceType();
}

4. Viva / Interview Defense Preparation

Q: "Why did you use BigDecimal instead of double for transaction amounts?"
Answer: double uses binary floating-point representation (IEEE-754) which cannot represent decimal fractions like 0.1 exactly (e.g. 0.1 + 0.2 = 0.30000000000000004). In financial reconciliation, accumulating floating point errors leads to false discrepancies. BigDecimal provides exact arbitrary-precision arithmetic and exact rounding control.
Q: "What OOP principles are represented in your design?"
Answer: 1. Abstraction: BaseFinancialRecord defines common contracts and the abstract method getSourceType().
2. Inheritance: OrderRecord, PaymentTransaction, and SettlementRecord inherit common fields using super().
3. Encapsulation: Private/protected fields with validation checks inside getters/setters (e.g., rejecting negative values).

Step 2: ReconciliationEngine.java (The O(N) Algorithm)

package com.paytrace.service;

import com.paytrace.model.*;
import java.math.BigDecimal;
import java.util.*;

public class ReconciliationEngine {
    private final BigDecimal tolerance;

    public ReconciliationEngine() {
        this(new BigDecimal("0.05")); // Default 5 cents tolerance
    }

    public ReconciliationEngine(BigDecimal tolerance) {
        this.tolerance = tolerance;
    }

    public ReconciliationResult reconcile(
            List<OrderRecord> orders,
            List<PaymentTransaction> payments,
            List<SettlementRecord> settlements
    ) {
        List<Discrepancy> discrepancies = new ArrayList<>();
        int matchedCount = 0;
        BigDecimal totalOrderValue = BigDecimal.ZERO;
        BigDecimal totalSettledValue = BigDecimal.ZERO;

        // 1. Index payments by orderId into Map<String, List<PaymentTransaction>> to detect duplicates
        Map<String, List<PaymentTransaction>> paymentMap = new HashMap<>();
        for (PaymentTransaction p : payments) {
            paymentMap.computeIfAbsent(p.getOrderId(), k -> new ArrayList<>()).add(p);
        }

        // 2. Index settlements by paymentId into Map<String, SettlementRecord>
        Map<String, SettlementRecord> settlementMap = new HashMap<>();
        for (SettlementRecord s : settlements) {
            settlementMap.put(s.getPaymentId(), s);
            totalSettledValue = totalSettledValue.add(s.getNetAmount());
        }

        // 3. 3-Way Match Loop over Orders
        int discIdCounter = 100;
        for (OrderRecord order : orders) {
            totalOrderValue = totalOrderValue.add(order.getAmount());
            String orderId = order.getRecordId();
            List<PaymentTransaction> matchingPayments = paymentMap.get(orderId);

            // Case A: Missing Payment
            if (matchingPayments == null || matchingPayments.isEmpty()) {
                discrepancies.add(new Discrepancy(
                    "DISC-" + (++discIdCounter), DiscrepancyType.MISSING_PAYMENT,
                    orderId, order.getAmount(), BigDecimal.ZERO,
                    "Order exists in OMS, but no payment gateway transaction was captured."
                ));
                continue;
            }

            // Case B: Duplicate Transactions
            if (matchingPayments.size() > 1) {
                BigDecimal totalPaid = matchingPayments.stream()
                        .map(PaymentTransaction::getAmount)
                        .reduce(BigDecimal.ZERO, BigDecimal::add);
                discrepancies.add(new Discrepancy(
                    "DISC-" + (++discIdCounter), DiscrepancyType.DUPLICATE_TRANSACTION,
                    orderId, order.getAmount(), totalPaid,
                    "Multiple payments (" + matchingPayments.size() + ") detected for single order #" + orderId
                ));
                continue;
            }

            PaymentTransaction payment = matchingPayments.get(0);

            // Case C: Amount Mismatch
            BigDecimal amountDiff = order.getAmount().subtract(payment.getAmount()).abs();
            if (amountDiff.compareTo(tolerance) > 0) {
                discrepancies.add(new Discrepancy(
                    "DISC-" + (++discIdCounter), DiscrepancyType.AMOUNT_MISMATCH,
                    orderId, order.getAmount(), payment.getAmount(),
                    "Order amount ($" + order.getAmount() + ") does not match gateway amount ($" + payment.getAmount() + ")."
                ));
                continue;
            }

            // Case D: Missing Settlement
            SettlementRecord settlement = settlementMap.get(payment.getRecordId());
            if (settlement == null) {
                discrepancies.add(new Discrepancy(
                    "DISC-" + (++discIdCounter), DiscrepancyType.MISSING_SETTLEMENT,
                    orderId, payment.getAmount(), BigDecimal.ZERO,
                    "Payment captured on gateway, but bank settlement record has not settled."
                ));
                continue;
            }

            matchedCount++;
        }

        return new ReconciliationResult(orders.size(), matchedCount, discrepancies, totalOrderValue, totalSettledValue);
    }
}

4. Viva / Interview Defense Preparation

Q: "How does your matching algorithm avoid an O(N^3) performance bottleneck?"
Answer: "If we used nested loops (checking every order against every payment and settlement), the time complexity would be $O(N \times M \times K)$, which for 2,000 records requires $8 \text{ billion comparisons}$ and causes the JVM to hang. By pre-indexing payments and settlements into in-memory HashMaps keyed by orderId and paymentId, lookups happen in constant $O(1)$ time, reducing overall algorithm complexity to linear $O(N)$."
Q: "Why did you use Map<String, List<PaymentTransaction>> instead of Map<String, PaymentTransaction>?"
Answer: "If a customer accidentally clicks 'Submit Payment' twice and gets double-charged, mapping to a single value would overwrite the first payment and hide the bug. Mapping to a List allows us to detect when the size is greater than 1 and flag a DUPLICATE_TRANSACTION."

package com.paytrace.ingestion;

import com.opencsv.CSVReader;
import com.paytrace.model.*;
import java.io.*;
import java.math.BigDecimal;
import java.time.LocalDateTime;

public class CsvImportService {

    public CsvValidationResult<OrderRecord> parseOrders(InputStream inputStream) {
        CsvValidationResult<OrderRecord> result = new CsvValidationResult<>();
        try (CSVReader reader = new CSVReader(new InputStreamReader(inputStream))) {
            String[] line;
            int lineNumber = 0;
            while ((line = reader.readNext()) != null) {
                lineNumber++;
                if (lineNumber == 1) continue; // Skip header

                if (line.length < 4) {
                    result.addWarning("Line " + lineNumber + ": Insufficient columns");
                    continue;
                }
                try {
                    String orderId = line[0].trim();
                    String customerId = line[2].trim();
                    BigDecimal amount = new BigDecimal(line[3].trim());
                    String currency = line.length > 4 ? line[4].trim() : "USD";
                    String status = line.length > 5 ? line[5].trim() : "COMPLETED";

                    if (orderId.isEmpty()) {
                        result.addWarning("Line " + lineNumber + ": Missing order_id");
                        continue;
                    }
                    result.addRecord(new OrderRecord(orderId, LocalDateTime.now(), customerId, amount, currency, status));
                } catch (NumberFormatException e) {
                    result.addWarning("Line " + lineNumber + ": Invalid numeric amount '" + line[3] + "'");
                } catch (Exception e) {
                    result.addWarning("Line " + lineNumber + ": Corrupted row - " + e.getMessage());
                }
            }
        } catch (Exception e) {
            result.addWarning("Fatal CSV read error: " + e.getMessage());
        }
        return result;
    }

    public CsvValidationResult<PaymentTransaction> parsePayments(InputStream inputStream) {
        CsvValidationResult<PaymentTransaction> result = new CsvValidationResult<>();
        try (CSVReader reader = new CSVReader(new InputStreamReader(inputStream))) {
            String[] line;
            int lineNumber = 0;
            while ((line = reader.readNext()) != null) {
                lineNumber++;
                if (lineNumber == 1) continue;
                try {
                    String paymentId = line[0].trim();
                    String orderId = line[1].trim();
                    BigDecimal amount = new BigDecimal(line[3].trim());
                    BigDecimal fee = line.length > 4 ? new BigDecimal(line[4].trim()) : BigDecimal.ZERO;
                    String status = line.length > 5 ? line[5].trim() : "CAPTURED";
                    String gateway = line.length > 6 ? line[6].trim() : "Stripe";

                    result.addRecord(new PaymentTransaction(paymentId, orderId, LocalDateTime.now(), amount, fee, gateway, status));
                } catch (Exception e) {
                    result.addWarning("Line " + lineNumber + ": Corrupted payment row");
                }
            }
        } catch (Exception e) {
            result.addWarning("Fatal CSV read error: " + e.getMessage());
        }
        return result;
    }

    public CsvValidationResult<SettlementRecord> parseSettlements(InputStream inputStream) {
        CsvValidationResult<SettlementRecord> result = new CsvValidationResult<>();
        try (CSVReader reader = new CSVReader(new InputStreamReader(inputStream))) {
            String[] line;
            int lineNumber = 0;
            while ((line = reader.readNext()) != null) {
                lineNumber++;
                if (lineNumber == 1) continue;
                try {
                    String settlementId = line[0].trim();
                    String paymentId = line[1].trim();
                    BigDecimal gross = new BigDecimal(line[3].trim());
                    BigDecimal net = new BigDecimal(line[4].trim());
                    BigDecimal fee = line.length > 5 ? new BigDecimal(line[5].trim()) : BigDecimal.ZERO;
                    String ref = line.length > 6 ? line[6].trim() : "BANK-REF";

                    result.addRecord(new SettlementRecord(settlementId, paymentId, LocalDateTime.now(), gross, net, fee, ref));
                } catch (Exception e) {
                    result.addWarning("Line " + lineNumber + ": Corrupted settlement row");
                }
            }
        } catch (Exception e) {
            result.addWarning("Fatal CSV read error: " + e.getMessage());
        }
        return result;
    }
}

Q: "How does your ingestion pipeline prevent a single corrupt row from failing an entire 10,000-row file?"
Answer: "I implemented fine-grained, row-level try-catch exception handling. Instead of throwing an unchecked exception that aborts the transaction, any NumberFormatException or missing field is isolated into a ValidationWarning log with the exact line number. The loop continues, preserving all valid records."
Q: "Why did you use InputStream instead of java.io.File?"
Answer: "In modern Spring Boot web applications, client uploads arrive via HTTP as MultipartFile, which streams input directly in memory as an InputStream. Using InputStream prevents costly disk I/O and works seamlessly for unit testing."

package com.paytrace.workflow;

import com.paytrace.model.Discrepancy;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;

public class InvestigationCase {
    private String caseId;
    private Discrepancy discrepancy;
    private InvestigationStatus status;
    private String assignedAnalyst;
    private String resolutionCategory;
    private String resolutionNote;
    private LocalDateTime createdAt;
    private LocalDateTime resolvedAt;
    private List<String> notes = new ArrayList<>();

    public InvestigationCase(String caseId, Discrepancy discrepancy, String assignedAnalyst) {
        this.caseId = caseId;
        this.discrepancy = discrepancy;
        this.assignedAnalyst = assignedAnalyst;
        this.status = InvestigationStatus.OPEN;
        this.createdAt = LocalDateTime.now();
    }

    public void moveToUnderReview() {
        if (this.status == InvestigationStatus.RESOLVED) {
            throw new IllegalStateException("Cannot move a RESOLVED case to UNDER_REVIEW directly. Reopen first.");
        }
        this.status = InvestigationStatus.UNDER_REVIEW;
    }

    public void resolve(String category, String note) {
        if (note == null || note.trim().isEmpty()) {
            throw new IllegalArgumentException("Resolution note is mandatory.");
        }
        this.status = InvestigationStatus.RESOLVED;
        this.resolutionCategory = category;
        this.resolutionNote = note;
        this.resolvedAt = LocalDateTime.now();
        this.notes.add("RESOLVED [" + category + "]: " + note);
    }

    public void reopen(String reason) {
        this.status = InvestigationStatus.REOPENED;
        this.notes.add("REOPENED: " + reason);
    }

    public void addNote(String author, String text) {
        this.notes.add(author + " (" + LocalDateTime.now() + "): " + text);
    }

    public String getCaseId() { return caseId; }
    public Discrepancy getDiscrepancy() { return discrepancy; }
    public InvestigationStatus getStatus() { return status; }
    public String getAssignedAnalyst() { return assignedAnalyst; }
    public List<String> getNotes() { return notes; }
    public String getResolutionCategory() { return resolutionCategory; }
}

        SettlementRecord settlement = new SettlementRecord("SET-2", "PAY-2", LocalDateTime.now(), new BigDecimal("230.00"), new BigDecimal("226.00"), new BigDecimal("4.00"), "BANK-2");

        ReconciliationResult result = engine.reconcile(List.of(order), List.of(payment), List.of(settlement));

        assertEquals(0, result.getMatchedRecords());
        assertEquals(1, result.getDiscrepanciesCount());
        assertEquals(DiscrepancyType.AMOUNT_MISMATCH, result.getDiscrepancies().get(0).getType());
        assertEquals(new BigDecimal("20.00"), result.getDiscrepancies().get(0).getVariance());
    }
